cmake_minimum_required(VERSION 3.24)
project(iprop_android LANGUAGES C CXX)
if(NOT IPROP_SOURCE)
    get_filename_component(IPROP_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../../../.." ABSOLUTE)
endif()
set(IPROP_RUNTIME_ONLY ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${IPROP_SOURCE}" iprop-native)
add_library(iprop_jni SHARED iprop_jni.cpp)
target_compile_features(iprop_jni PRIVATE cxx_std_20)
target_link_libraries(iprop_jni PRIVATE iprop::capi_static)
iprop_warnings(iprop_jni)
if(NOT ANDROID)
    find_package(JNI REQUIRED)
    target_include_directories(iprop_jni PRIVATE ${JNI_INCLUDE_DIRS})
endif()
if(ANDROID)
    # Compatible with Android devices using 16 KiB pages, including older NDKs.
    target_link_options(iprop_jni PRIVATE "-Wl,-z,max-page-size=16384")
endif()
