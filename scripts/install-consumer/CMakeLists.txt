cmake_minimum_required(VERSION 3.24)
project(iprop_install_consumer LANGUAGES C)
find_package(iprop 0.2 CONFIG REQUIRED)
add_executable(consumer main.c)
set_target_properties(consumer PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
if(USE_STATIC)
    enable_language(CXX)
    target_link_libraries(consumer PRIVATE iprop::capi_static)
    set_target_properties(consumer PROPERTIES LINKER_LANGUAGE CXX)
else()
    target_link_libraries(consumer PRIVATE iprop::capi)
endif()
if(MSVC)
    target_compile_options(consumer PRIVATE /W4 /WX)
    target_compile_definitions(consumer PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
