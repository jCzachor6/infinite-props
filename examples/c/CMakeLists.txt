cmake_minimum_required(VERSION 3.24)
project(parameter_panel LANGUAGES C CXX)
find_package(iprop 0.4 CONFIG REQUIRED)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
foreach(linkage shared static)
    add_executable(parameter_panel_${linkage} parameter_panel.c)
    if(linkage STREQUAL "shared")
        target_link_libraries(parameter_panel_${linkage} PRIVATE iprop::capi)
    else()
        target_link_libraries(parameter_panel_${linkage} PRIVATE iprop::capi_static)
        set_target_properties(parameter_panel_${linkage} PROPERTIES LINKER_LANGUAGE CXX)
    endif()
    if(MSVC)
        target_compile_options(parameter_panel_${linkage} PRIVATE /W4 /WX /utf-8)
        target_compile_definitions(parameter_panel_${linkage} PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(parameter_panel_${linkage} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
