cmake_minimum_required(VERSION 3.24)
project(iprop_c_examples LANGUAGES C CXX)
find_package(iprop 0.4 CONFIG REQUIRED)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
foreach(example parameter_panel render_inputs)
    foreach(linkage shared static)
        set(target ${example}_${linkage})
        add_executable(${target} ${example}.c)
        if(linkage STREQUAL "shared")
            target_link_libraries(${target} PRIVATE iprop::capi)
        else()
            target_link_libraries(${target} PRIVATE iprop::capi_static)
            set_target_properties(${target} PROPERTIES LINKER_LANGUAGE CXX)
        endif()
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
            target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        endif()
    endforeach()
endforeach()
